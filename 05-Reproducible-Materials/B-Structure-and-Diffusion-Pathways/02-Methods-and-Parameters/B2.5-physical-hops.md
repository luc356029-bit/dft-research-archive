### 1.目的
[4.cutoff距离得到总扩散路径条数N](B-analysis.md#2方案)

### 2.输入 、内化 → 实践强化
**材料层：**
1. hop为LiA→LiB的连接，可表示为（i，j，R）。
其中 i 表示 `Li i` （索引）；j 表示 `Li j` ; R表示 `Li j` 所在的晶胞相对于 `Li i` 所在晶胞的三维坐标位置。
如果是物理hop，当去除方向性，即 LiA-LiB
2. 同一物理hop LiA-LiB 包括了 LiA→LiB 和 LiB→LiA ，被枚举了两次，我们可以令（i，j，R）= (j，i，-R)，
两条方向相反的hop因此合并为一条物理hop
3. 需要划定Li-Li hop的枚举范围，对于一个hop来说，起始两个Li，应该至少有一个在晶胞内

**代码层：**
1. 对于hop (i，i，R)，当R为（0，0，0）时，此hop相当于自己连接自己，不合法；R≠（0，0，0）时，合法
2. N与枚举范围相关，如果枚举范围从晶胞变成超胞，N也会相应变化，因此一整个复刻过程中，都不能随意更改枚举范围
### 3.实践
**环境配置、文件配置：**
无
**运行、对比：**
1. 对LiAl进行验证
2. 重点检查hop是否两两成对，N是否为所有hop数的一半。LiAl检查没问题
3. 对Li2Al进行验证，通过
4. Li2Al的结果比较有意思：Li2Al的Li原子更多，但物理hop数和LiAl相同。
这说明影响到几何扩散路径条数的不仅仅是原子数，至少还包括了晶体的结构拓扑
LiAl在d0时形成了3d扩散路径，而Li2Al只形成了1d。LiAl的Li少，但每个Li分叉多，Li-Li之间的连接度高（x，y，z）三个方向都有；而Li2Al只在y轴上形成hop

**源码**
```python
from pymatgen.core import Structure

# =========================
# 参数
# =========================
D0 = 2.7395798528   # LiAl，来自 B2.4，Å
EPS = 1e-7


def canonical_hop(i, j, image):
    """
    将同一无方向物理 hop：
        (i, j, R)
        (j, i, -R)
    统一成唯一 canonical key。
    """

    rx, ry, rz = (int(x) for x in image)

    forward = (
        int(i), int(j),
        rx, ry, rz
    )

    reverse = (
        int(j), int(i),
        -rx, -ry, -rz
    )

    # 固定保留字典序较小的一种表示
    return min(forward, reverse)


def analyze_hops(cif_path):
    structure = Structure.from_file(cif_path)

    query_cutoff = D0 + EPS

    print("=" * 70)
    print("文件:", cif_path)
    print("化学式:", structure.composition.reduced_formula)
    print(f"d0 = {D0:.10f} Å")
    print(f"程序搜索 cutoff = {query_cutoff:.10f} Å")

    # 当前晶胞内全部 Li
    li_indices = [
        i for i, site in enumerate(structure)
        if site.species_string == "Li"
    ]

    print("\n晶胞内全部 Li indices:", li_indices)

    raw_hops = []
    unique_hops = {}

    # ==================================================
    # 1. 从当前晶胞内每一个 Li 出发，枚举 d0 内 Li 邻居
    # ==================================================
    for i in li_indices:

        center = structure[i]

        neighbors = structure.get_neighbors(
            center,
            query_cutoff
        )

        for neighbor in neighbors:

            if neighbor.species_string != "Li":
                continue

            j = int(neighbor.index)
            image = tuple(int(x) for x in neighbor.image)
            distance = float(neighbor.nn_distance)

            # 排除真正的自己 → 自己
            if (
                i == j
                and image == (0, 0, 0)
            ):
                continue

            key = canonical_hop(
                i,
                j,
                image
            )

            raw_hops.append(
                {
                    "i": i,
                    "j": j,
                    "image": image,
                    "distance": distance,
                    "key": key
                }
            )

            # canonical key 相同 → 同一物理 hop
            if key not in unique_hops:
                unique_hops[key] = distance

    # ==================================================
    # 2. 打印原始枚举
    # ==================================================
    print("\n=== 原始有方向 hop ===")

    for n, hop in enumerate(raw_hops, start=1):
        print(
            f"{n}: "
            f"Li{hop['i']} (0,0,0) -> "
            f"Li{hop['j']} {hop['image']} | "
            f"d={hop['distance']:.10f} Å | "
            f"canonical={hop['key']}"
        )

    # ==================================================
    # 3. 打印 canonical 去重后的物理 hop
    # ==================================================
    print("\n=== 去重后的无方向物理 hop ===")

    for n, key in enumerate(
        sorted(unique_hops),
        start=1
    ):
        i, j, rx, ry, rz = key

        print(
            f"{n}: "
            f"(Li{i}, Li{j}, "
            f"R=({rx},{ry},{rz})) | "
            f"d={unique_hops[key]:.10f} Å"
        )

    raw_N = len(raw_hops)
    N = len(unique_hops)

    # ==================================================
    # 4. 基本合理性检查
    # ==================================================
    print("\n=== B2.5 结果 ===")
    print("原始有方向 hop 数 =", raw_N)
    print("去重后物理 hop 数 N =", N)

    print(
        "正反向成对检查:",
        raw_N == 2 * N
    )


analyze_hops(
    "cif_files/mp-1067_LiAl.cif"
)
```

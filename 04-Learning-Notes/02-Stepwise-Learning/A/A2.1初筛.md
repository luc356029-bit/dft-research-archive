**源码：**
```python
# import环节，导入需要的工具
from mp_api.client import MPRester
import os
import csv

API_KEY = os.environ["MP_API_KEY"]

# 列出不合条件的元素
exclude_elements = {
    # 稀贵金属
    "Au","Ag","Pt","Pd","Rh","Ir","Ru","Os","Re",
    # 有毒元素
    "Hg","Pb","Cd","Tl",
    # 放射性元素
    "U","Th","Ra","Po","Tc","Np","Pu",
    # 碱金属（Li本身除外）
    "Na","K","Rb","Cs","Fr",
    # 碱土金属
    "Be","Mg","Ca","Sr","Ba",
    # 典型非金属（不包括准金属Si/Ge/Sb）
    "H","C","N","O","F","P","S","Cl","Se","Br","I","At","Te","As",
    # 稀有气体
    "He","Ne","Ar","Kr","Xe","Rn",
}

# 从mp中批量导入可能要用的二元锂合金材料
with MPRester(API_KEY) as mpr:
    docs = mpr.materials.summary.search(
        elements=["Li"],
        num_elements=2,
        fields=[
            "material_id",
            "formula_pretty",
            "structure",
            "energy_above_hull",
            "band_gap",
            "elements"
        ]
    )

print(f"原始查询到：{len(docs)} 条记录")

# 筛选：排除元素 + Ehull（单位：eV/atom）
filtered = []
for doc in docs:
    other_elements = [str(e) for e in doc.elements if str(e) != "Li"]
    if any(el in exclude_elements for el in other_elements):
        continue
    if doc.energy_above_hull >= 0.05:
        continue
    filtered.append(doc)

print(f"元素+Ehull过滤后：{len(filtered)} 条记录")

# 保存全部合格结构，同化学式的不同结构按各自MP-ID分别保存
os.makedirs("cif_files", exist_ok=True)
os.makedirs("summary", exist_ok=True)

with open("summary/alloy_list.csv", "w", newline="") as f:
    writer = csv.writer(f)
    writer.writerow(["mp_id", "formula", "ehull", "band_gap"])
    for doc in filtered:
        cif_path = f"cif_files/{doc.material_id}_{doc.formula_pretty}.cif" 
        doc.structure.to(filename=cif_path)
        writer.writerow([
            doc.material_id,
            doc.formula_pretty,
            doc.energy_above_hull,
            doc.band_gap
        ])
        print(f"已保存：{doc.formula_pretty} ({doc.material_id})")

print("完成。")
```
